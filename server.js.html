const express = require('express');
const cors = require('cors');
const { Pool } = require('pg');
const app = express();
app.use(cors());
app.use(express.json());

// Connect to PostgreSQL (set DATABASE_URL on Render)
const pool = new Pool({
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false }
});

// Fetch businesses by province, town, category
app.get('/api/businesses', async (req, res) => {
    const { province, town, category } = req.query;
    if(!province || !town || !category)
        return res.status(400).json({error:'Missing parameters'});

    const query = `
        SELECT b.name, b.address, b.phone, t.name AS town, p.name AS province, c.name AS category
        FROM businesses b
        JOIN towns t ON b.town_id = t.id
        JOIN provinces p ON t.province_id = p.id
        JOIN categories c ON b.category_id = c.id
        WHERE p.name=$1 AND t.name=$2 AND c.name=$3
    `;

    try {
        const result = await pool.query(query, [province, town, category]);
        res.json(result.rows);
    } catch(err) {
        console.error(err);
        res.status(500).json({error: err.message});
    }
});

app.listen(3000, () => console.log('Backend running on port 3000'));
